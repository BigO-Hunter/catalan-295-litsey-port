/-- The original deformed balanced prefix block is a unit. Its two
node families are independent actual nodes, not a supplied pivot matrix. -/
theorem balanced_unit (q o k j0 gamma m N b : ℕ) (u : GaussLoc p)
    (hq0 : 0<q) (hqp : q<p) (hq2 : 2≤q) (hoq : o<q)
    (hk0 : 0<k) (hk : k≤p) (hg : gamma<p) (hroot : q*k=j0*p+o)
    (hsize : 2*m≤N) (hN : q*N<p*p) (hb : b≤m/p+1) (hb0 : 1≤b)
    (x y : Fin b → ℕ) (hx : Function.Injective x) (hy : Function.Injective y)
    (hxN : ∀ i,gamma+x i*p<N) (hyN : ∀ i,gamma+y i*p<N)
    (hxy : ∀ i j,y j≠x i) (hu : res p u≠0) :
    IsUnit (Matrix.of fun i j => WrapDeformedEntry.entry p q o u
      (gamma+x i*p) (gamma+y j*p)).det := by
  classical
  have hp := (Fact.out : p.Prime).pos
  have hquot (j : ℕ) : (gamma+j*p)/p=j := by
    rw [Nat.add_mul_div_right _ _ hp,Nat.div_eq_of_lt hg,zero_add]
  have hxb : ∀ i,x i+b<p := by
    intro i
    have hh := WrapPrefixBounds.quotient_add_count p q m N (gamma+x i*p) b
      (Fact.out : p.Prime).one_lt hq2 hsize hN (hxN i) hb
    simpa only [hquot] using hh
  have hyb : ∀ i,y i+b<p := by
    intro i
    have hh := WrapPrefixBounds.quotient_add_count p q m N (gamma+y i*p) b
      (Fact.out : p.Prime).one_lt hq2 hsize hN (hyN i) hb
    simpa only [hquot] using hh
  have hbp : b<p := by have := hxb ⟨0,hb0⟩; omega
  have hsmallx : ∀ i,x i<p := by intro i; have := hxb i; omega
  have hsmally : ∀ i,y i<p := by intro i; have := hyb i; omega
  let point := WrapTestPoint.point p q j0 b (if k≤gamma then 1 else 0)
  let Q := Matrix.of fun i j => WrapDeformedEntry.entry p q o u (gamma+x i*p) (gamma+y j*p)
  let T := Matrix.of fun i j => WrapEntryTest.value p q o point u (gamma+x i*p) (gamma+y j*p)
  have ht : T.map (res p)=res p u • WrapHarmonic.matrix (K:=RatFunc (ZMod p)) b (decide (k≤gamma)) x y := by
    apply Matrix.ext; intro i j
    exact WrapOwnEntry.residue p q o k j0 gamma m N b (x i) (y j) u hq0 hqp hq2 hoq
      hk0 hk hg hroot hsize hN (hxN i) (hyN j) hb hb0 (hsmallx i) (hsmally j) (Ne.symm (hxy i j))
  have hfp := WrapHarmonicPivot.det_ne p b (decide (k≤gamma)) x y hbp hx hy hxb hyb hxy
  have hrf : (WrapHarmonic.matrix (K:=RatFunc (ZMod p)) b (decide (k≤gamma)) x y).det≠0 := by
    rw [← WrapHarmonicMap.matrix (RatFunc.C:ZMod p →+* RatFunc (ZMod p))]
    change ((RatFunc.C:ZMod p →+* RatFunc (ZMod p)).mapMatrix _).det≠0
    rw [← RingHom.map_det]
    exact fun hz => hfp ((RatFunc.C:ZMod p →+* RatFunc (ZMod p)).injective
      (by simpa only [map_zero] using hz))
  have htest : res p T.det≠0 := by
    rw [RingHom.map_det (R:=GaussLoc p) (S:=RatFunc (ZMod p)) (res p) T]
    change (T.map (res p)).det≠0
    rw [ht]
    exact WrapScale.det_ne _ hu _ hrf
  have he : ∀ i j,WrapPoleTest.At p point (Q i j) (T i j) := by
    intro i j
    by_cases hle : x i≤y j
    · exact WrapOwnTest.entry_at p q o k j0 gamma (y j) m N b hq0 hqp hq2 hoq hk0 hk hg hroot
        hsize hN (hyN j) hb hb0 u _ _
        (Nat.add_le_add_left (Nat.mul_le_mul_right p hle) gamma) le_rfl
    · have hle' : y j≤x i := by omega
      exact WrapOwnTest.entry_at p q o k j0 gamma (x i) m N b hq0 hqp hq2 hoq hk0 hk hg hroot
        hsize hN (hxN i) hb hb0 u _ _ le_rfl
        (Nat.add_le_add_left (Nat.mul_le_mul_right p hle') gamma)
  have hd := det_relation p point Q T he
  have hgeneric := transfer p (piL p) (res_eq_zero_iff p) (WrapDeform.pi_prime p) hd htest
  change IsUnit Q.det
  exact unit_ne p Q.det hgeneric
