variable (p : ℕ) [Fact p.Prime]

theorem q_ne (q : ℕ) (hq : res p (q:GaussLoc p)≠0) :
    (q:WrapPoleResidue.Residue p)≠0 := by
  have hh : RatFunc.C (res p (q:GaussLoc p))=(q:WrapPoleResidue.Residue p) := by
    rw [WrapSimplePole.res_nat]
    simp only [map_natCast]
  intro hz
  apply hq
  apply (RatFunc.C:RatFunc (ZMod p) →+* WrapPoleResidue.Residue p).injective
  simpa only [map_zero] using hh.trans hz

/-- The actual deformed cumulative load has the original pole-by-pole
harmonic form, including every absorbed full period. -/
theorem residue_load (q o k j0 gamma j N : ℕ)
    (hq0 : 0<q) (hqp : q<p) (hoq : o<q) (hk0 : 0<k) (hk : k≤p)
    (hg : gamma<p) (hroot : q*k=j0*p+o) (hN : N<p*p) (hnode : gamma+j*p<N) :
    WrapDeform.residue p (∑ t∈Icc 1 (gamma+j*p),WrapIncrement.delta p q o t)=
      WrapHarmonic.load ((j0:WrapPoleResidue.Residue p)/(q:WrapPoleResidue.Residue p)+RatFunc.X)
        (decide (k≤gamma)) j := by
  have hp := (Fact.out : p.Prime).pos
  have hqb := WrapSimplePole.q_res_ne p q hq0 hqp
  have hqf : (q:ZMod p)≠0 := by
    intro hz
    apply hqb
    exact (WrapSimplePole.res_nat_zero p q).mpr ((ZMod.natCast_eq_zero_iff _ _).mp hz)
  have hh : WrapDeform.residue p (∑ t∈Icc 1 (gamma+j*p),WrapIncrement.delta p q o t)=
      WrapPoleLoad.pure p (gamma+j*p)-WrapPoleLoad.fractional p q o k
        (RatFunc.X:WrapPoleResidue.Residue p) (gamma+j*p) := by
    rw [map_sum]
    unfold WrapPoleLoad.pure WrapPoleLoad.fractional
    rw [← Finset.sum_sub_distrib]
    apply Finset.sum_congr rfl
    intro t ht
    have ht1 := (Finset.mem_Icc.mp ht).1
    have htN : t<N := lt_of_le_of_lt (Finset.mem_Icc.mp ht).2 hnode
    have ht2 : ¬p^2∣t := Smith.not_sq_dvd_of_lt ht1 (lt_trans htN hN)
    have hle : o≤q*t := by have := Nat.le_mul_of_pos_right q ht1; omega
    rw [WrapPoleResidue.delta p q o t hqb hle ht2]
    have hiff := WrapPoleLoad.fractional_pole_iff p q o k j0 t hoq ht1 hroot hqf
    by_cases hd : p∣q*t-o
    · rw [if_pos hd,if_pos (hiff.mp hd)]
      simp only [one_div,div_eq_mul_inv,one_mul]
    · rw [if_neg hd,if_neg (fun h => hd (hiff.mpr h)),mul_zero]
  exact hh.trans (WrapPoleLoad.load_eq p q o k j0 gamma j
    (RatFunc.X:WrapPoleResidue.Residue p) hp hk0 hk hg hroot (q_ne p q hqb))
