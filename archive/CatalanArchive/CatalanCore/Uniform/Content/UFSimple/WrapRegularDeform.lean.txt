variable (p : ℕ) [Fact p.Prime]

noncomputable def constants := (WrapDeform.embed p).comp
  (Polynomial.C:GaussLoc p →+* (GaussLoc p)[X])

/-- Moving the generic fractional pole leaves every original nonpole
normalized reciprocal unchanged modulo pi squared. -/
theorem normalized_congr (a b : GaussLoc p) (ha : res p a≠0) :
    WrapDeform.uniformizer p^2∣WrapPole.normalizedInverse p a b-
      constants p (WrapPole.originalNormalized p a) := by
  classical
  have hu := isUnit_of_res_ne a ha
  have hA : constants p a*constants p (Ring.inverse a)=1 := by
    rw [← map_mul,Ring.mul_inverse_cancel _ hu,map_one]
  have hD := Ring.mul_inverse_cancel _ (WrapPole.denominator_unit p a b ha)
  have he : WrapPole.denominator p a b=constants p a+
      WrapDeform.uniformizer p*constants p b*WrapDeform.embed p Polynomial.X := by
    change WrapDeform.embed p (Polynomial.C a+Polynomial.C (piL p)*Polynomial.C b*Polynomial.X)=
      WrapDeform.embed p (Polynomial.C a)+WrapDeform.embed p (Polynomial.C (piL p))*
        WrapDeform.embed p (Polynomial.C b)*WrapDeform.embed p Polynomial.X
    simp only [map_add,map_mul]
  have hD' : (constants p a+WrapDeform.uniformizer p*constants p b*
      WrapDeform.embed p Polynomial.X)*Ring.inverse (WrapPole.denominator p a b)=1 := by
    rw [← he]
    exact hD
  have hi : Ring.inverse (WrapPole.denominator p a b)-constants p (Ring.inverse a)=
      -WrapDeform.uniformizer p*constants p b*WrapDeform.embed p Polynomial.X*
        Ring.inverse (WrapPole.denominator p a b)*constants p (Ring.inverse a) := by
    linear_combination -Ring.inverse (WrapPole.denominator p a b)*hA+
      constants p (Ring.inverse a)*hD'
  rw [WrapPole.normalizedInverse,dif_neg ha,WrapPole.originalNormalized,dif_neg ha,map_mul]
  change WrapDeform.uniformizer p^2∣WrapDeform.uniformizer p*
    Ring.inverse (WrapPole.denominator p a b)-
      WrapDeform.uniformizer p*constants p (Ring.inverse a)
  refine ⟨-constants p b*WrapDeform.embed p Polynomial.X*
    Ring.inverse (WrapPole.denominator p a b)*constants p (Ring.inverse a),?_⟩
  calc
    _=WrapDeform.uniformizer p*(Ring.inverse (WrapPole.denominator p a b)-
      constants p (Ring.inverse a)) := by ring
    _=_ := by rw [hi]; ring

/-- The complete normalized two-term increment retains its original
first-order residue at each original fractional nonpole. -/
theorem delta_congr (q o t : ℕ) (ha : res p (WrapIncrement.argument p q o t)≠0) :
    WrapDeform.uniformizer p^2∣WrapIncrement.delta p q o t-
      constants p (WrapIncrement.originalDelta p q o t) := by
  obtain ⟨z,hz⟩ := normalized_congr p (WrapIncrement.argument p q o t) (q:GaussLoc p) ha
  refine ⟨-constants p (q:GaussLoc p)*z,?_⟩
  unfold WrapIncrement.delta WrapIncrement.originalDelta
  rw [map_sub,map_mul]
  change constants p (WrapPole.originalNormalized p (t:GaussLoc p))-
    constants p (q:GaussLoc p)*WrapPole.normalizedInverse p (WrapIncrement.argument p q o t) (q:GaussLoc p)-
    (constants p (WrapPole.originalNormalized p (t:GaussLoc p))-
      constants p (q:GaussLoc p)*constants p (WrapPole.originalNormalized p (WrapIncrement.argument p q o t)))=_
  linear_combination -constants p (q:GaussLoc p)*hz
