/-- A mixed spectral core over any domain embeds faithfully in its
fraction field. The balanced block may be polynomial in the pole parameter. -/
theorem mixed_det_ne_zero {R F ι κ : Type*} [CommRing R] [IsDomain R] [Field F]
    [Fintype ι] [DecidableEq ι] [Fintype κ] [DecidableEq κ]
    (f : R →+* F) (hf : Function.Injective f)
    (A : Matrix ι ι R) (B : Matrix ι κ R) (E : Matrix κ ι R) (D : Matrix κ κ R)
    (hD : D.det≠0) :
    (Matrix.fromBlocks ((X:R[X]) • (1:Matrix ι ι R[X])+A.map C)
      (B.map C) (E.map C) (D.map C)).det≠0 := by
  let g := Polynomial.mapRingHom f
  have hDf : (D.map f).det≠0 := by
    change (f.mapMatrix D).det≠0
    rw [← RingHom.map_det]
    exact fun hz => hD (hf (by simpa only [map_zero] using hz))
  have hh := WrapPivot.mixed_det_ne_zero (A.map f) (B.map f) (E.map f) (D.map f) hDf
  let M := Matrix.fromBlocks ((X:R[X]) • (1:Matrix ι ι R[X])+A.map C)
    (B.map C) (E.map C) (D.map C)
  have hm : M.map g=Matrix.fromBlocks ((X:F[X]) • (1:Matrix ι ι F[X])+(A.map f).map C)
      ((B.map f).map C) ((E.map f).map C) ((D.map f).map C) := by
    apply Matrix.ext; intro i j
    rcases i with i|i <;> rcases j with j|j
    · by_cases hij : i=j <;>
        simp [M,g,Matrix.one_apply,hij,Polynomial.map_add,Polynomial.map_C,Polynomial.map_X]
    · simp [M,g,Polynomial.map_C]
    · simp [M,g,Polynomial.map_C]
    · simp [M,g,Polynomial.map_C]
  intro hz
  apply hh
  rw [← hm]
  change (g.mapMatrix M).det=0
  rw [← RingHom.map_det]
  rw [show M.det=0 from hz,map_zero]
