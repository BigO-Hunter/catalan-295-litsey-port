variable {K ι : Type*} [Field K] [Fintype ι] [DecidableEq ι]

/-- An arrow determinant clears the reference normalization without
evaluating an inverse matrix at the test point. -/
def arrow (Q : Matrix ι ι K) (sigma : ι → K) : Matrix (ι ⊕ Unit) (ι ⊕ Unit) K :=
  Matrix.fromBlocks Q (Matrix.of fun _ _ => 1) (Matrix.of fun _ j => sigma j) (1:Matrix Unit Unit K)

theorem det_arrow (Q : Matrix ι ι K) (hQ : IsUnit Q.det) (sigma : ι → K) :
    (arrow Q sigma).det=Q.det*(1-sigma ⬝ᵥ (Q⁻¹ *ᵥ 1)) := by
  letI : Invertible Q := Matrix.invertibleOfIsUnitDet Q hQ
  unfold arrow
  rw [Matrix.det_fromBlocks₁₁,Matrix.invOf_eq_nonsing_inv,Matrix.det_unique (n:=Unit)]
  simp only [Matrix.sub_apply,Matrix.one_apply_eq,Matrix.mul_apply,
    Matrix.of_apply,Matrix.mulVec,dotProduct,Pi.one_apply,mul_one,Finset.sum_mul,Finset.mul_sum]
  rw [Finset.sum_comm]

theorem det_arrow_ne (Q : Matrix ι ι K) (hQ : Q.det≠0) (sigma : ι → K)
    (href : 1-sigma ⬝ᵥ (Q⁻¹ *ᵥ 1)≠0) : (arrow Q sigma).det≠0 := by
  rw [det_arrow Q (isUnit_iff_ne_zero.mpr hQ) sigma]
  exact mul_ne_zero hQ href

/-- Append the cleared balanced reference to the constant block before
adding the original common-node spectral diagonal. -/
theorem spectral_arrow_ne {κ : Type*} [Fintype κ] [DecidableEq κ]
    (A : Matrix κ κ K) (B : Matrix κ (ι ⊕ Unit) K)
    (E : Matrix (ι ⊕ Unit) κ K) (D : Matrix ι ι K) (sigma : ι → K)
    (hD : D.det≠0) (href : 1-sigma ⬝ᵥ (D⁻¹ *ᵥ 1)≠0) :
    (Matrix.fromBlocks ((X:K[X]) • (1:Matrix κ κ K[X])+A.map C)
      (B.map C) (E.map C) ((arrow D sigma).map C)).det≠0 :=
  WrapPivot.mixed_det_ne_zero A B E (arrow D sigma) (det_arrow_ne D hD sigma href)

end CatalanCore.Uniform.Content.UFSimple.WrapArrow
