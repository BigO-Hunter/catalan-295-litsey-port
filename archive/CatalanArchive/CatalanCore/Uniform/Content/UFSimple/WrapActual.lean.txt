common-node coordinates. -/

    (htlo : r+2*l-1≤t) (hthi : t<r+p)
    (a : {a : ι // ((c a:ℕ):ZMod p) = ((t:ℕ):ZMod p)}) :
    rowCoords c r s l hl hls (Sum.inl ((surplus_fibre_equiv c r s l t hl hls hp htlo hthi).symm a).1) = c a.1 ∧
