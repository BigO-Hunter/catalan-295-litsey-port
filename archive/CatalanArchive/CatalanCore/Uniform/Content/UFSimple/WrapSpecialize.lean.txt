variable {R K L ι : Type*} [CommRing R] [Field K] [Field L]
  [Fintype ι] [DecidableEq ι]

/-- The reference numerator is polynomial even when the inverse is not. -/
noncomputable def numerator (Q : Matrix ι ι R) (sigma : ι → R) : R :=
  Q.det-sigma ⬝ᵥ (Q.adjugate *ᵥ 1)

theorem map_numerator (f : R →+* K) (Q : Matrix ι ι R) (sigma : ι → R) :
    f (numerator Q sigma)=numerator (Q.map f) (f ∘ sigma) := by
  have ha : Q.adjugate.map f=(Q.map f).adjugate := f.map_adjugate Q
  unfold numerator
  simp only [Matrix.mulVec,dotProduct,Pi.one_apply,mul_one,map_sub,map_sum,map_mul,
    RingHom.map_det,Function.comp_def]
  rw [← ha]
  rfl

/-- Clearing the inverse requires only the determinant unit of this matrix. -/
theorem numerator_eq (Q : Matrix ι ι K) (sigma : ι → K) (hQ : IsUnit Q.det) :
    numerator Q sigma=Q.det*(1-sigma ⬝ᵥ (Q⁻¹ *ᵥ 1)) := by
  have hh := Matrix.det_smul_inv_mulVec_eq_cramer Q (1:ι→K) hQ
  rw [Matrix.cramer_eq_adjugate_mulVec] at hh
  unfold numerator
  rw [← hh,dotProduct_smul]
  simp only [smul_eq_mul]
  ring

theorem numerator_ne (Q : Matrix ι ι K) (sigma : ι → K) (hQ : Q.det≠0)
    (hB : 1-sigma ⬝ᵥ (Q⁻¹ *ᵥ 1)≠0) : numerator Q sigma≠0 := by
  rw [numerator_eq Q sigma (isUnit_iff_ne_zero.mpr hQ)]
  exact mul_ne_zero hQ hB

/-- A single valid test value proves nonvanishing in an injective generic
field. No evaluation of a rational function inverse is involved. -/
theorem transfer (test : R →+* K) (generic : R →+* L)
    (hinj : Function.Injective generic) (Q : Matrix ι ι R) (sigma : ι → R)
    (hQ : (Q.map test).det≠0)
    (hB : 1-(test ∘ sigma) ⬝ᵥ ((Q.map test)⁻¹ *ᵥ 1)≠0) :
    (Q.map generic).det≠0 ∧
      1-(generic ∘ sigma) ⬝ᵥ ((Q.map generic)⁻¹ *ᵥ 1)≠0 := by
  have hd : Q.det≠0 := by
    intro hz
    apply hQ
    change (test.mapMatrix Q).det=0
    rw [← RingHom.map_det,hz,map_zero]
  have hn : numerator Q sigma≠0 := by
    intro hz
    have ht := numerator_ne (Q.map test) (test ∘ sigma) hQ hB
    apply ht
    rw [← map_numerator,hz,map_zero]
  have hgd : (Q.map generic).det≠0 := by
    change (generic.mapMatrix Q).det≠0
    rw [← RingHom.map_det]
    exact fun hz => hd (hinj (by simpa only [map_zero] using hz))
  refine ⟨hgd,?_⟩
  intro hz
  apply hn
  apply hinj
  rw [map_zero,map_numerator,numerator_eq _ _ (isUnit_iff_ne_zero.mpr hgd),hz,mul_zero]

end CatalanCore.Uniform.Content.UFSimple.WrapSpecialize
